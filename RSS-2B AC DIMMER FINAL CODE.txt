int AC_LOAD = 3;
int dimming = 128;

  77,75,73,70,68,65,61,57,53,49,
  46,43,40,37,35,34,32,30,28,27,45
};

unsigned long startupDelay = 50;
unsigned long startupTime;

void ss_logic(int status);

/* ================= UART ================= */

void UART_send(char data) {
    while (!(UCSR0A & (1 << UDRE0)));
    UDR0 = data;
}

void UART_sendString(const char *str) {
    while (*str) {
        UART_send(*str++);
    }
}

void UART_Begin() {
    UBRR0 = 103;
    UCSR0C |= (1 << UCSZ01) | (1 << UCSZ00);
    UCSR0B |= (1 << RXEN0) | (1 << TXEN0) | (1 << RXCIE0);
}

/* ================= ZERO CROSS ================= */

void zero_crosss_int()
{
    if ((millis() - startupTime) < startupDelay) return;

    int dimtime = (75 * dimming);
    delayMicroseconds(dimtime);

    digitalWrite(AC_LOAD, HIGH);
    delayMicroseconds(10);
    digitalWrite(AC_LOAD, LOW);
}

/* ================= SETUP ================= */

void setup()
{
    UART_Begin();
    pinMode(AC_LOAD, OUTPUT);
    digitalWrite(AC_LOAD, LOW);
    sei();

    dimming = 128;

    // Optional debug
    UART_sendString("Setup done.\n");
}

/* ================= UART ISR ================= */

ISR(USART_RX_vect)
{
    char data = UDR0;

    if (data == '#') {
        idx = 0;  // restart frame
    }

    if (idx < sizeof(temp) - 1) {
        temp[idx++] = data;
    }
}

/* ================= LOOP ================= */

void loop()
{
    if (idx >= 4)   // we only care about first 4 chars
    {
        delay(50);  // allow full command

        char val[3] = {0};
        memcpy(val, &temp[1], 2);

        initchar = temp[0];
        modechar = temp[3];

        count = atoi(val);

        idx = 0;
        memset(temp, 0, sizeof(temp));

        // ❌ NO ERR CHECKS → silently ignore invalid

        /* ================= MODE I ================= */
        if (initchar == '#' && (modechar == 'I' || modechar == 'i'))
        {
            if (count != 0)
            {
                startupTime = millis();
                attachInterrupt(0, zero_crosss_int, RISING);
                sei();
                status = 1;
            }
            else
            {
                // TURN OFF
                EIMSK &= ~(1 << INT0);
                dimming = 128;
                digitalWrite(AC_LOAD, LOW);
                status = 1;

                // ✅ ONLY HERE SEND OK
                UART_sendString("OK\n");
            }
        }

        /* ================= MODE C ================= */
        else if (initchar == '#' && (modechar == 'C' || modechar == 'c'))
        {
            if (count != 0)
            {
                if (count <= 30)
                {
                    dimming = dimmingValues[count - 1];
                }
                attachInterrupt(0, zero_crosss_int, RISING);
                status = 0;
            }
            else
            {
                // TURN OFF
                EIMSK &= ~(1 << INT0);
                dimming = 128;
                digitalWrite(AC_LOAD, LOW);
                status = 0;

                // ✅ ONLY HERE SEND OK
                UART_sendString("OK\n");
            }
        }
    }

    ss_logic(status);
}
